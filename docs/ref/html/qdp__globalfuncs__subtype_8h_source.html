148template<class T>
149typename UnaryReturn<OLattice<T>, FnSum>::Type_t
150sum( const OSubLattice<T>& s1 )
151{
152 typename UnaryReturn<OLattice<T>, FnSum>::Type_t d;
153
154 // Must initialize to zero since we do not know if the loop will be entered
155 zero_rep(d.elem());
156
157 for(int j=0; j < s1.subset().numSiteTable(); ++j)
158 {
159 d.elem() += s1.getF()[j];
160 }
161
162 // Do a global sum on the result
163 QDPInternal::globalSum(d);
164
165 return d;
166 }
