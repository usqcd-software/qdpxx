60/*
61 Generate a table for a byte-wise 32-bit CRC calculation on the polynomial:
62 x^32+x^26+x^23+x^22+x^16+x^12+x^11+x^10+x^8+x^7+x^5+x^4+x^2+x+1.
63
64 Polynomials over GF(2) are represented in binary, one bit per coefficient,
65 with the lowest powers in the most significant bit. Then adding polynomials
66 is just exclusive-or, and multiplying a polynomial by x is a right shift by
67 one. If we call the above polynomial p, and represent a byte as the
68 polynomial q, also with the lowest power in the most significant bit (so the
69 byte 0xb1 is the polynomial x^7+x^3+x+1), then the CRC is (q*x^32) mod p,
70 where a mod b means the remainder after dividing a by b.
71
72 This calculation is done using the shift-register method of multiplying and
73 taking the remainder. The register is initialized to zero, and for each
74 incoming bit, x^32 is added mod p to the register if the bit is a one (where
75 x^32 mod p is p+x^32 = x^26+...+1), and the register is multiplied mod p by
76 x (which is shifting right by one and adding x^32 mod p if the bit shifted
77 out is a one). We start with the highest power (least significant bit) of
78 q and repeat for all eight bits of q.
79
80 The table is simply the CRC of all possible eight bit values. This is all
81 the information needed to generate CRC's on data a byte at a time for all
82 combinations of CRC register values and incoming bytes.
83*/
84 local void make_crc_table()
85 {
86 uLong c;
87 int n, k;
88 uLong poly; /* polynomial exclusive-or pattern */
89 /* terms of polynomial defining this crc (except x^32): */
90 static Byte p[] = {0,1,2,4,5,7,8,10,11,12,16,22,23,26};
91
92 /* make exclusive-or pattern from polynomial (0xedb88320L) */
93 poly = 0L;
94 for (n = 0; n < sizeof(p)/sizeof(Byte); n++)
95 poly |= 1L << (31 - p[n]);
96
97 for (n = 0; n < 256; n++)
98 {
99 c = (uLong)n;
100 for (k = 0; k < 8; k++)
101 c = c & 1 ? poly ^ (c >> 1) : c >> 1;
102 crc_table[n] = c;
103 }
104 crc_table_empty = 0;
105 }

177/* ========================================================================= */
178#define DO1(buf) crc = crc_table[((int)crc ^ (*buf++)) & 0xff] ^ (crc >> 8);
179#define DO2(buf) DO1(buf); DO1(buf);
180#define DO4(buf) DO2(buf); DO2(buf);
181#define DO8(buf) DO4(buf); DO4(buf);
182
183/* ========================================================================= */
184 n_uint32_t crc32(n_uint32_t crc, const unsigned char *buf, size_t len)
185 {
186 if (buf == Z_NULL) return 0L;
187#ifdef DYNAMIC_CRC_TABLE
188 if (crc_table_empty)
189 make_crc_table();
190#endif
191 crc = crc ^ 0xffffffffL;
192 while (len >= 8)
193 {
194 DO8(buf);
195 len -= 8;
196 }
197 if (len)
198 {
199 do {
200 DO1(buf);
201 } while (--len);
202 }
203 return crc ^ 0xffffffffL;
204 }
