74void ArrayMap::make(const ArrayMapFunc& func)
75{
76 // We are allowed to declare a mapsa, but not allocate one.
77 // There is an empty constructor for Map. Hence, the resize will
78 // actually allocate the space.
79 mapsa.resize(func.numArray());
80
81 // Loop over each direction making the Map
82 for(int dir=0; dir < func.numArray(); ++dir)
83 {
84 PackageArrayMapFunc my_local_map(func,dir);

86 mapsa[dir].make(my_local_map);
