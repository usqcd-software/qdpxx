51 void TimeSliceIO<T>::setTimeSlice(int time_slice)
52 {
53 current_time = time_slice;
54 int tDir = Nd-1;
55
56 if ((current_time < 0) || (current_time >= Layout::lattSize()[tDir]))
57 {
58 throw(std::string("Invalid time in Lattice TimeSliceIO"));
59 }
60
61 multi1d<int> coord(Nd);
62 coord = 0;
63 coord[tDir] = current_time;
64 start_lexico = QDP::local_site(coord,Layout::lattSize());
65 if (current_time == Layout::lattSize()[tDir]-1)
66 stop_lexico = Layout::vol();
67 else
68 {
69 coord[tDir]++;
70 stop_lexico = QDP::local_site(coord,Layout::lattSize());
71 }
72 }
