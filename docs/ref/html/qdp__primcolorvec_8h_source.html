204template<class T1, class T2, int N>
205inline typename BinaryReturn<PColorVector<T1,N>, PColorVector<T2,N>, FnLocalColorInnerProduct>::Type_t
206localColorInnerProduct(const PColorVector<T1,N>& l, const PColorVector<T2,N>& r)
207{
208 typename BinaryReturn<PColorVector<T1,N>, PColorVector<T2,N>, FnLocalColorInnerProduct>::Type_t d;
209
210 // The traceColor is eaten here
211 d.elem() = adj(l.elem(0)) * r.elem(0);
212 for(int k=1; k < N; ++k)
213 d.elem() += adj(l.elem(k)) * r.elem(k);
214
215 return d;
216}

264template<class T1, class T2, class T3>
265inline typename TrinaryReturn<PColorVector<T1,3>, PColorVector<T2,3>, PColorVector<T3,3>, FnColorContract>::Type_t
266colorContract(const PColorVector<T1,3>& s1, const PColorVector<T2,3>& s2, const PColorVector<T3,3>& s3)
267{
268 typename TrinaryReturn<PColorVector<T1,3>, PColorVector<T2,3>, PColorVector<T3,3>, FnColorContract>::Type_t d;
269
270 // Permutations: +(0,1,2)+(1,2,0)+(2,0,1)-(1,0,2)-(0,2,1)-(2,1,0)
271
272 // d = \epsilon^{i,j,k} V1^{i} V2^{j} V3^{k}
273 d.elem() = (s1.elem(0)*s2.elem(1)
274 - s1.elem(1)*s2.elem(0))*s3.elem(2)
275 + (s1.elem(1)*s2.elem(2)
276 - s1.elem(2)*s2.elem(1))*s3.elem(0)
277 + (s1.elem(2)*s2.elem(0)
278 - s1.elem(0)*s2.elem(2))*s3.elem(1);
279
280 return d;
281}

299colorVectorContract(const PColorVector<T1,N>& s1, const PColorVector<T2,N>& s2)
